```rust
/// 列出脚本运行历史：读 script-runs.json 全量或按 script_id 过滤。
/// 仅终态（运行态查询走 script_status），输出含 output_tail（M2-4.d 已落盘）。
#[tauri::command]
pub fn script_runs_list(
    app: AppHandle,
    webview: tauri::Webview,
    script_id: Option<String>,
) -> Result<Vec<ScriptRunRecord>, String> {
    check_invocation_source(&webview, "script_runs_list", None, &app)?;
    let path = workspace::script_runs_file(&app);
    let mut records = script_runner::load_run_records(&path);
    if let Some(sid) = script_id.as_deref() {
        records.retain(|r| r.script_id == sid);
    }
    workspace::log_audit(
        &app,
        "script.runs.list",
        format!(
            "count={} script_id={}",
            records.len(),
            script_id.as_deref().unwrap_or("*")
        ),
    );
    Ok(records)
}
```
