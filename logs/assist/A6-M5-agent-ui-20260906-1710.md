### 2.1 Skill 侧类型

```ts
// —— 执行描述（镜像 A5 §3.1 SkillExec；禁 Inline，K6）——
export type SkillExec =
  | { kind: "script_ref"; scriptId: string; params: Record<string, unknown> }   // 复用 M2-3 已存脚本
  | { kind: "command_ref"; commandId: string; params: Record<string, unknown> } // 复用 M2-3 已存命令
  | { kind: "sequence"; steps: SkillExec[] };                                    // 串联，每步递归走上述二者

// —— ACL 等级（镜像 A5 §3.2 AclLevel）——
export type AclLevel = "safe" | "confirm" | "dangerous";

export type SkillSource = "builtin" | "local" | "registry";
export type SkillParamType = "string" | "number" | "bool" | "path" | "enum" | "password";
export type SkillOutputKind = "text" | "json" | "markdown" | "file";

export interface SkillParam {
  name: string; label: string; type: SkillParamType;
  required: boolean; default?: string;
  options?: { value: string; label: string }[];
  pattern?: string; dangerous: boolean; dangerReason?: string;
}
export interface SkillMeta {
  id: string; name: string; description: string; version: string;
  author?: string; source: SkillSource;
  inputs: SkillParam[];
  outputs: { type: SkillOutputKind; schema?: unknown };
  capabilities: string[];                 // 引用 capability.rs 常量（CapabilityRef）
  acl: AclLevel;                           // 安装/运行闸门等级（取代 W1 草案游离 acl 推断）
  exec: SkillExec;                         // 执行描述（取代 W1 草案 implementation.kind 枚举，对齐 K6）
  timeoutSec: number; dangerous: boolean; tags: string[];
}
export interface SkillInstall {
  meta: SkillMeta; enabled: boolean;
  installedAt: string; updatedAt: string;
  grantedCapabilities: string[];           // 实际授予的能力（⊆ whitelist）
  checksum: string; origin?: string;
  grantedScopes?: Record<string, string[]>; // 运行期内存持有，不落盘（K3）
}
export interface SkillRunRecord {
  runId: string; skillId: string; version: string;
  status: RunStatus; startedAt: string; finishedAt?: string;
  exitCode?: number; durationMs?: number; truncated: boolean;
  error?: { code: string; message: string; retriable: boolean; details?: string };
  // 参数值脱敏后镜像（password → "***"），输出正文不进前端持久状态
}
```

### 2.3 共用：能力引用 + 运行态 + 流式载荷

```ts
// —— 能力引用（capability.rs 单源；M5-3/M5-2 共用一份，防漂移）——
export type CapabilityRef = string; // 如 "fs:read" | "net:http" | "skill:invoke:<id>"

// —— 运行态（复用 M4 RunStatus：domain.rs:667；扩展 Agent 周期触发语义）——
export type RunStatus =
  | "running" | "succeeded" | "failed" | "cancelled" | "timeout"
  | "missed" | "retry" | "queued"; // missed/retry/queued 供 Agent 周期触发回显

// —— 流式载荷（W3 修订：Tauri event，非 Channel；结构同构 TermMessage）——
// 后端经 Tauri event 推：agent://<agent_id>/stream（Agent，A5 §3.4 已定）
//                           skill://<run_id>/stream（Skill，M5-4 §4.3，A16 落地时定 topic）
// 节流：≥50ms 批量 + 累计 ≤16KB 强制 flush；心跳每 5s；终止 .../done|.../error|.../canceled
export type StreamChunk =
  | { kind: "data"; runId?: string; sessionId?: string; data: string }
  | { kind: "flow"; runId?: string; sessionId?: string; droppedChunks?: number; droppedBytes?: number }
  | { kind: "exit"; runId?: string; sessionId?: string; error?: { code: string; message: string } };
```

> - §2.1 `SkillMeta.implementation.kind` → 改为 `SkillMeta.exec: SkillExec`（引用而非内嵌，对齐 K6 禁 Inline；取代 W1 草案 `SkillImplKind`）。
> - §2.1 新增 `SkillMeta.acl: AclLevel`（W1 草案未显式建模 ACL 等级，靠 `dangerous` 推断；现与 A5 §3.2 对齐）。
> - §2.3 流式：W1 §4.1 推荐 `Channel` 单播 → **W3 改为 Tauri event**（`agent://.../stream` / `skill://.../stream`），与 A5 §3.4（A1 卡已定 `app.emit` 方案）一致。前端 `bridge.ts` 收 event → `useAgentStore` 累积（见 §3.2）。
